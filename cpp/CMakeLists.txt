cmake_minimum_required(VERSION 3.20)
# version: single source in include/piazzi/version.h
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/include/piazzi/version.h PIAZZI_VERSION_LINE REGEX "#define PIAZZI_VERSION ")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" PIAZZI_VERSION "${PIAZZI_VERSION_LINE}")
project(piazzi VERSION ${PIAZZI_VERSION} LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# No external dependencies: the core builds the same on macOS, Windows, Linux and WebAssembly (Emscripten).
file(GLOB CORE_SRC CONFIGURE_DEPENDS src/*.cpp)
list(FILTER CORE_SRC EXCLUDE REGEX ".*main_.*\\.cpp$")
add_library(piazzi_core STATIC ${CORE_SRC})
target_include_directories(piazzi_core PUBLIC include)
if(NOT EMSCRIPTEN)
  add_executable(piazzi src/main_cli.cpp)
  target_link_libraries(piazzi piazzi_core)

# ---------------------------------------------------------------- GUI app (Dear ImGui + SDL2 + OpenGL3/WebGL2)
option(ASTRO_GUI "Build the GUI app" ON)
if(ASTRO_GUI)
  include(FetchContent)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.93.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(imgui)
  set(IMGUI_SRC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
      ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  file(GLOB GUI_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/gui/*.cpp)
  set(GUI_ASSETS_DIR ${CMAKE_SOURCE_DIR}/gui/assets)
  file(GLOB_RECURSE GUI_ASSETS CONFIGURE_DEPENDS ${GUI_ASSETS_DIR}/*)
  list(FILTER GUI_ASSETS EXCLUDE REGEX ".*/\\.gitkeep$")
  add_executable(piazzi_app ${GUI_SRC} ${IMGUI_SRC})
  target_include_directories(piazzi_app PRIVATE ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(piazzi_app piazzi_core)
  if(EMSCRIPTEN)
    set_target_properties(piazzi_app PROPERTIES OUTPUT_NAME "index" SUFFIX ".html")
    target_compile_options(piazzi_app PRIVATE -sUSE_SDL=2)
    target_link_options(piazzi_app PRIVATE -sUSE_SDL=2 -sUSE_WEBGL2=1 -sFULL_ES3=1 -sALLOW_MEMORY_GROWTH=1 -sFETCH=1
        -sEXPORTED_RUNTIME_METHODS=FS -sEXPORTED_FUNCTIONS=_main,_astro_file_added --shell-file ${CMAKE_SOURCE_DIR}/gui/shell.html)
  else()
    # ASTRO_FETCH_SDL2=ON: builds SDL2 statically from source (used by the packaging/ scripts).
    option(ASTRO_FETCH_SDL2 "Download and build SDL2 statically (self-contained app)" OFF)
    if(ASTRO_FETCH_SDL2)
      enable_language(C)
      set(SDL_SHARED OFF CACHE BOOL "" FORCE)
      set(SDL_STATIC ON CACHE BOOL "" FORCE)
      set(SDL_TEST OFF CACHE BOOL "" FORCE)
      set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
      FetchContent_Declare(SDL2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.30.9/SDL2-2.30.9.tar.gz)
      FetchContent_MakeAvailable(SDL2)
      if(TARGET SDL2::SDL2main)
        target_link_libraries(piazzi_app SDL2::SDL2main)
      endif()
      target_link_libraries(piazzi_app SDL2::SDL2-static)
    else()
      find_package(SDL2 REQUIRED)
      if(TARGET SDL2::SDL2main)
        target_link_libraries(piazzi_app SDL2::SDL2main)
      endif()
      target_link_libraries(piazzi_app SDL2::SDL2)
    endif()
    find_package(OpenGL REQUIRED)
    target_link_libraries(piazzi_app OpenGL::GL)
    if(APPLE)
      set_source_files_properties(${GUI_ASSETS} PROPERTIES HEADER_FILE_ONLY TRUE)
      foreach(a ${GUI_ASSETS})
        file(RELATIVE_PATH rel ${GUI_ASSETS_DIR} ${a})
        get_filename_component(reldir ${rel} DIRECTORY)
        set_source_files_properties(${a} PROPERTIES MACOSX_PACKAGE_LOCATION Resources/assets/${reldir})
      endforeach()
      target_sources(piazzi_app PRIVATE ${GUI_ASSETS})
      # bundle icon (packaging/icon/piazzi.icns in Contents/Resources)
      set(PIAZZI_ICNS ${CMAKE_SOURCE_DIR}/../packaging/icon/piazzi.icns)
      set_source_files_properties(${PIAZZI_ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
      target_sources(piazzi_app PRIVATE ${PIAZZI_ICNS})
      set_target_properties(piazzi_app PROPERTIES MACOSX_BUNDLE TRUE OUTPUT_NAME "Piazzi"
          MACOSX_BUNDLE_BUNDLE_NAME "Piazzi" MACOSX_BUNDLE_ICON_FILE "piazzi.icns" MACOSX_BUNDLE_GUI_IDENTIFIER "ai.datamonkeys.piazzi"
          MACOSX_BUNDLE_SHORT_VERSION_STRING "${PIAZZI_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "${PIAZZI_VERSION}")
    else()
      # assets next to the executable
      add_custom_command(TARGET piazzi_app POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E copy_directory ${GUI_ASSETS_DIR} $<TARGET_FILE_DIR:piazzi_app>/assets)
    endif()
    if(WIN32)
      set_target_properties(piazzi_app PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME "Piazzi")
      # executable icon: .rc resource with packaging/icon/piazzi.ico (windres on MinGW, rc on MSVC)
      enable_language(RC)
      target_sources(piazzi_app PRIVATE ${CMAKE_SOURCE_DIR}/../packaging/icon/piazzi.rc)
      set_property(SOURCE ${CMAKE_SOURCE_DIR}/../packaging/icon/piazzi.rc APPEND PROPERTY INCLUDE_DIRECTORIES
          ${CMAKE_SOURCE_DIR}/../packaging/icon)
      if(MINGW)
        target_link_options(piazzi_app PRIVATE -static -static-libgcc -static-libstdc++ -mwindows)
      endif()
    endif()
  endif()
endif()

# ---------------------------------------------------------------- tests (ctest). They run from the repository root,
# since they read cpp/gui/assets/example and cpp/tests/data by relative path.
if(NOT EMSCRIPTEN)
  enable_testing()
  get_filename_component(ASTRO_REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/.. ABSOLUTE)
  foreach(t test_detect test_measure test_display test_orbits test_image_io test_api_robust test_accuracy test_psf_fit test_linking test_stack test_photometry test_centroid test_trail)
    add_test(NAME ${t} COMMAND ${t} WORKING_DIRECTORY ${ASTRO_REPO_ROOT})
  endforeach()
  # end-to-end shell tests: POSIX shell only (on Windows the CLI is covered by test_accuracy and the same core tests)
  if(NOT WIN32)
    add_test(NAME accuracy_cli COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/accuracy.sh ${CMAKE_BINARY_DIR} WORKING_DIRECTORY ${ASTRO_REPO_ROOT})
    find_program(PYTHON3 python3)
    if(PYTHON3)
      add_test(NAME report_format COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/report_format.sh ${CMAKE_BINARY_DIR} WORKING_DIRECTORY ${ASTRO_REPO_ROOT})
    endif()
  endif()
endif()
